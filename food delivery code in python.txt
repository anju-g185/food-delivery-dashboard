import pandas as pd
import random
from datetime import datetime, timedelta

restaurants = [
    ("R001", "Spice Hub", "Indian"),
    ("R002", "Pizza Point", "Italian"),
    ("R003", "Biryani House", "Biryani"),
    ("R004", "Burger Town", "Fast Food"),
    ("R005", "Dosa Corner", "South Indian")
]

cities = ["Mumbai", "Pune", "Nashik", "Nagpur", "Thane"]

payment_methods = ["UPI", "Card", "Cash", "Wallet"]

statuses = ["Delivered", "Delivered", "Delivered",
            "Delivered", "Cancelled"]

data = []

start_date = datetime(2026, 1, 1)

for i in range(1, 1001):

    restaurant = random.choice(restaurants)

    order_date = start_date + timedelta(
        days=random.randint(0, 179)
    )

    order_value = random.randint(150, 1500)
    delivery_fee = random.randint(20, 80)
    discount = random.choice([0, 20, 50, 100, 150])

    status = random.choice(statuses)

    if status == "Delivered":
        delivery_time = random.randint(15, 60)
        rating = round(random.uniform(3, 5), 1)
    else:
        delivery_time = 0
        rating = 0

    row = [
        f"ORD{i:04d}",
        order_date.strftime("%Y-%m-%d"),
        f"C{random.randint(1, 200):03d}",
        restaurant[0],
        restaurant[1],
        random.choice(cities),
        restaurant[2],
        order_value,
        delivery_fee,
        discount,
        random.choice(payment_methods),
        status,
        delivery_time,
        rating,
        round(random.uniform(1, 15), 1)
    ]

    data.append(row)

columns = [
    "Order_ID",
    "Order_Date",
    "Customer_ID",
    "Restaurant_ID",
    "Restaurant_Name",
    "City",
    "Cuisine",
    "Order_Value",
    "Delivery_Fee",
    "Discount",
    "Payment_Method",
    "Order_Status",
    "Delivery_Time_Min",
    "Customer_Rating",
    "Distance_KM"
]

df = pd.DataFrame(data, columns=columns)

df.to_csv("food_delivery.csv", index=False)

print(df.head())
print(df.shape)

df["Net_Revenue"] = (
    df["Order_Value"]
    + df["Delivery_Fee"]
    - df["Discount"]
)
